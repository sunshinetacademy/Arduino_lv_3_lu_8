توقّع أولاً

ما الذي يجعل نظاماً يحتوي على 4 حساسات + OLED + Buzzer + RTC يعمل بشكل منظم — لا فوضى؟

✅ الإجابة:

State Machine (آلة الحالات)! بدلاً من كتابة كود خطي عشوائي، نُقسّم سلوك النظام إلى حالات واضحة:

🟦 IDLE — النظام يراقب بهدوء
🟨 DETECTED — تم اكتشاف حركة
🟥 ALERT — إطلاق الإنذار
🟪 COOLDOWN — فترة راحة قبل إعادة المراقبة

النظام لا ينتقل لأي حالة إلا وفق شرط. هذا يجعل الكود نظيفاً وسهل التطوير.

⚡ خريطة الجهود — انتبه للخلط!

⚠️ مكونات 3.3V

• ESP32 (اللوحة كلها)
• RTC DS3231 (VCC)
• BME280 (VCC)
• OLED SSD1306 (VCC)

5V = حرق فوري!

⚠️ مكونات 5V

• PIR HC-SR501 (VCC)
• Buzzer (يمكن 3.3V أو 5V)

3.3V قد لا تكفي PIR!

⚠️ تحذير 3 — فحص العناوين:

قبل التركيب، استخدم I2C Scanner للتأكد من عناوين المكونات. لو ظهر تعارض (نفس العنوان)، غيّر العنوان من مكتبة الإعدادات.

⚠️ تحذير 4 — Buzzer صاخب:

Buzzer قد يكون صوتاً عالياً. ضع مقاومة 100Ω على التوالي لتقليل شدة الصوت — أو استخدم tone() بترددات متغيّرة.

💡 نصيحة المشاريع: اختبر كل مكون على حدة أولاً (كما فعلنا في L3-02 حتى L3-06)، ثم اجمعهم واحداً واحداً. لا تبدأ بالكود الكامل من أول مرة!

بدلاً من كود خطي فوضوضي، نُقسّم سلوك النظام إلى حالات. النظام ينتقل من حالة لحالة وفق شروط واضحة.

🟦 IDLE

النظام يراقب. إذا كشف PIR حركة → انتقل لـ DETECTED. إذا دخل أمر من Blynk → تنفيذ ثم عودة.

🟨 DETECTED

تم كشف حركة! سجّل الوقت من RTC، اعرض على OLED → انتقل لـ ALERT مباشرة.

🟥 ALERT

شغّل Buzzer + LED. بعد 5 ثوانٍ → COOLDOWN.

🟪 COOLDOWN

انتظر 10 ثوانٍ (لتجنب تكرار سريع). بعدها عُد لـ IDLE.

🎯 لماذا State Machine؟ الكود يصبح: • أسهل في القراءة • أسهل في التصحيح • أسهل في التطوير (أضف حالة = أضف قطعة صغيرة).

🧪 الكود 1: اختبار Buzzer

07 / 11

قبل الدمج، نتأكد أن Buzzer يعمل — ونتعلم كيف نولّد نغمات مختلفة.

// اختبار Buzzer — نغمات مختلفة
#define BUZZER_PIN 26

void setup() {
  Serial.begin(115200);
  pinMode(BUZZER_PIN, OUTPUT);
  Serial.println("Buzzer Test");
}

void loop() {
  // نغمة تنبيه قصيرة (1 kHz)
  tone(BUZZER_PIN, 1000);
  delay(200);
  noTone(BUZZER_PIN);
  delay(100);

  // نغمة إنذار عالية (2 kHz)
  tone(BUZZER_PIN, 2000);
  delay(100);
  noTone(BUZZER_PIN);
  delay(100);

  // نغمة منخفضة (500 Hz)
  tone(BUZZER_PIN, 500);
  delay(300);
  noTone(BUZZER_PIN);
  delay(1000);
}
✅ نتائج متوقعة: 3 نغمات مختلفة تسمعها بشكل واضح
💡 tone(pin, hz): التردد بالهرتز — كلما زاد، زاد حِدّة الصوت
⚠️ لو الصوت خافت: بعض Buzzers تحتاج 5V. استخدم طرف VIN بدلاً من 3.3V — لكن فقط لـ Buzzer!

🛡️ الكود 2: النظام المتكامل

08 / 11
📝 قبل الكود: هذا الكود يشغّل 3 مكونات I2C + 2 Digital + State Machine. اقرأه بعناية — مركّب من 4 أجزاء: Setup + Helper Functions + State Machine + Loop. 
#include <Wire.h>
#include <RTClib.h>
#include <Adafruit_BME280.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

// ═══ Pin Definitions ═══
#define PIR_PIN     27
#define BUZZER_PIN  26
#define LED_PIN     25
#define SCREEN_W    128
#define SCREEN_H    64
#define OLED_ADDR   0x3C

// ═══ Timing Constants ═══
#define ALERT_DURATION_MS    5000   // 5 ثواني إنذار
#define COOLDOWN_DURATION_MS 10000  // 10 ثواني راحة

// ═══ Objects ═══
RTC_DS3231        rtc;
Adafruit_BME280    bme;
Adafruit_SSD1306   display(SCREEN_W, SCREEN_H, &Wire, -1);

// ═══ State Machine ═══
enum SystemState {
  STATE_IDLE,
  STATE_DETECTED,
  STATE_ALERT,
  STATE_COOLDOWN
};
SystemState currentState = STATE_IDLE;
unsigned long stateStartTime = 0;
int lastPir = LOW;
int eventCount = 0;
char lastEventTime[16] = "--:--:--";

// ═══ Helper: عرض الشاشة ═══
void updateDisplay() {
  DateTime now = rtc.now();
  float t = bme.readTemperature();

  display.clearDisplay();
  display.setTextColor(SSD1306_WHITE);

  // السطر 1: الحالة
  display.setTextSize(1);
  display.setCursor(0, 0);
  switch (currentState) {
    case STATE_IDLE:
      display.println("* Room Guard - IDLE");
      break;
    case STATE_DETECTED:
      display.println("! DETECTED !");
      break;
    case STATE_ALERT:
      display.println(">> ALERT <<");
      break;
    case STATE_COOLDOWN:
      display.println(".. Cooldown ..");
      break;
  }

  // السطر 2: الوقت
  display.setTextSize(2);
  display.setCursor(0, 16);
  char buf[16];
  sprintf(buf, "%02d:%02d:%02d",
          now.hour(), now.minute(), now.second());
  display.println(buf);

  // السطر 3: آخر حدث
  display.setTextSize(1);
  display.setCursor(0, 36);
  display.print("Last: ");
  display.println(lastEventTime);

  // السطر 4: عدد الأحداث + الحرارة
  display.setCursor(0, 50);
  display.print("Events: ");
  display.print(eventCount);
  display.print(" T:");
  display.print(t, 1);
  display.println("C");

  display.display();
}

// ═══ Helper: تغيير الحالة ═══
void setState(SystemState newState) {
  currentState = newState;
  stateStartTime = millis();
  Serial.print("[STATE] -> ");
  Serial.println(newState);
}

// ═══ Setup ═══
void setup() {
  Serial.begin(115200);
  pinMode(PIR_PIN, INPUT);
  pinMode(BUZZER_PIN, OUTPUT);
  pinMode(LED_PIN, OUTPUT);

  if (!rtc.begin()) Serial.println("RTC error");
  if (!bme.begin(0x76)) Serial.println("BME error");

  display.begin(SSD1306_SWITCHCAPVCC, OLED_ADDR);
  display.clearDisplay();
  display.setTextSize(1);
  display.setCursor(0, 20);
  display.println("  Room Guard");
  display.setCursor(0, 36);
  display.println("  Starting...");
  display.display();

  delay(2000);  // PIR warm-up
  setState(STATE_IDLE);
}

// ═══ Loop — State Machine ═══
void loop() {
  int pir = digitalRead(PIR_PIN);
  unsigned long elapsed = millis() - stateStartTime;

  switch (currentState) {
    // ───── IDLE: مراقبة ─────
    case STATE_IDLE:
      digitalWrite(LED_PIN, LOW);
      if (pir == HIGH && lastPir == LOW) {
        setState(STATE_DETECTED);
      }
      break;

    // ───── DETECTED: تسجيل + تحضير ─────
    case STATE_DETECTED: {
      DateTime now = rtc.now();
      sprintf(lastEventTime, "%02d:%02d:%02d",
              now.hour(), now.minute(), now.second());
      eventCount++;
      Serial.print("EVENT #");
      Serial.print(eventCount);
      Serial.print(" at ");
      Serial.println(lastEventTime);
      setState(STATE_ALERT);
      break;
    }

    // ───── ALERT: إنذار ─────
    case STATE_ALERT:
      tone(BUZZER_PIN, 1500);
      digitalWrite(LED_PIN, HIGH);
      if (elapsed > ALERT_DURATION_MS) {
        noTone(BUZZER_PIN);
        digitalWrite(LED_PIN, LOW);
        setState(STATE_COOLDOWN);
      }
      break;

    // ───── COOLDOWN: راحة ─────
    case STATE_COOLDOWN:
      if (elapsed > COOLDOWN_DURATION_MS) {
        setState(STATE_IDLE);
      }
      break;
  }

  lastPir = pir;

  // تحديث الشاشة كل 500ms فقط (لتجنب الوميض)
  static unsigned long lastDisplay = 0;
  if (millis() - lastDisplay > 500) {
    updateDisplay();
    lastDisplay = millis();
  }
}
🎯 5 نقاط ذكية في الكود:

1. enum SystemState — تعريف واضح للحالات الأربعة.
2. switch/case — كل حالة لها منطقها الخاص.
3. millis() بدل delay — لا يوقف باقي الكود.
4. static unsigned long — متغير يُحفظ بين استدعاءات loop.
5. تحديث OLED كل 500ms — يمنع الوميض ويحسّن الأداء.
